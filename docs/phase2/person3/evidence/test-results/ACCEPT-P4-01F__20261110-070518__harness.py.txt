from pathlib import Path
import hashlib, re, subprocess, datetime, os

ROOT = Path(r'C:\Users\DELL\source\repos\CSDLNC-DA')
EV = ROOT / 'docs/phase2/person3/evidence/test-results'
UTC = datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%d-%H%M%S')
LOG = EV / f'ACCEPT-P4-01F__{UTC}.txt'
results = []
MASK = [m for m in {os.environ.get('COMPUTERNAME', '')} if len(m) > 2]


def hide(s):
    for m in MASK:
        s = re.sub(re.escape(m), '<local-server>', s, flags=re.I)
    return s


def log(s):
    with LOG.open('a', encoding='utf-8') as f:
        f.write(hide(str(s)) + '\n')


def check(name, ok, detail=''):
    results.append((name, bool(ok)))
    line = f'{"PASS" if ok else "FAIL"} | {name} | {detail}'
    log(line)
    print(hide(line), flush=True)


def sql(label, db, args, quiet=False):
    p = subprocess.run(['sqlcmd', '-S', 'localhost', '-E', '-I', '-b', '-l', '10', '-t', '600', '-d', db, '-W', '-s', '|'] + args,
                       cwd=ROOT, capture_output=True)
    txt = p.stdout.decode('utf-8', errors='replace') + p.stderr.decode('utf-8', errors='replace')
    if not quiet:
        log('\n== ' + label + ' ==\n' + txt)
    if p.returncode:
        raise RuntimeError(f'{label}: sqlcmd exit {p.returncode}: {txt[-800:]}')
    return txt

log('Scope: evidence validity only. No functional test re-run (directive section 2: "Do not rerun completed tests unnecessarily"), '

log('Branch: ' + subprocess.check_output(['git', 'branch', '--show-current'], cwd=ROOT, text=True).strip())

        'E4 T-REC-05 regression data incl. spending cross-check': ('T-REC-05__20261110-0206_regression.txt', r'T-REC-05_RESULT: PASS', r'T-REC-05 SUMMARY[^\n]*'),
